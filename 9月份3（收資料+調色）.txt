        function copyToClipboard(text) {
            const tempInput = document.createElement('textarea');
            tempInput.value = text;
            tempInput.style.position = 'fixed';
            tempInput.style.opacity = '0';
            document.body.appendChild(tempInput);
            tempInput.focus();
            tempInput.select();
            try {
                document.execCommand('copy');
            } catch (err) {
                console.error('Copy failed', err);
            }
            document.body.removeChild(tempInput);
        }

        function handleSubmit() {
            const selectedCourses = Array.from(document.querySelectorAll('input[name="course"]:checked')).map(el => el.value);

            if (selectedCourses.length === 0) {
                showToast('請至少勾選一場欲報名的講座場次！', 'error');
                return;
            }

            const name = document.getElementById('name').value.trim();
            const lineName = document.getElementById('lineName').value.trim();
            const actualAmount = document.getElementById('actualAmount').value.trim();
            const bankLast5 = document.getElementById('bankLast5').value.trim();

            if (!name || !lineName || !actualAmount || !bankLast5) {
                showToast('請填寫所有必填欄位 (*)', 'error');
                return;
            }

            const needReceipt = document.getElementById('needReceipt').checked;
            const receiptTitle = document.getElementById('receiptTitle').value.trim();
            const receiptTaxId = document.getElementById('receiptTaxId').value.trim();

            let receiptInfo = "不需開立收據";
            if (needReceipt) {
                receiptInfo = `需要收據 (抬頭: ${receiptTitle || '無'}${receiptTaxId ? ' / 統編: ' + receiptTaxId : ''})`;
            }

            const isHsinchu = document.getElementById('isHsinchu').checked ? "是（現場報到領取首場半價 $150 優惠金）" : "否";

            const summaryText = `【9月FLOW微講座報名核對】\n姓名：${name}\nLINE 暱稱：${lineName}\n新竹縣市民：${isHsinchu}\n收據需求：${receiptInfo}\n----------------\n報名場次：\n- ${selectedCourses.join('\n- ')}\n----------------\n系統試算總額：NT$ ${document.getElementById('totalPrice').innerText}\n實際填寫匯款：NT$ ${actualAmount}\n匯款後五碼：${bankLast5}\n（請檢附繳費憑證照片，以核對帳目資料）`;

            copyToClipboard(summaryText);

            const submitBtn = document.getElementById('submitBtn');
            submitBtn.disabled = true;
            submitBtn.innerHTML = '<i class="fa-solid fa-spinner fa-spin text-xl sm:text-2xl"></i> 資料傳送中...';

            const payload = {
                name: name,
                email: document.getElementById('email').value.trim() || '未提供',
                lineName: lineName,
                isHsinchu: isHsinchu,
                receiptInfo: receiptInfo,
                courses: selectedCourses.join(' / '),
                totalPrice: document.getElementById('totalPrice').innerText,
                actualAmount: actualAmount,
                bankLast5: bankLast5
            };

            fetch(SCRIPT_URL, {
                method: 'POST',
                mode: 'no-cors',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            }).then(() => {

                submitBtn.innerHTML = '<i class="fa-solid fa-paper-plane text-xl sm:text-2xl"></i> 確認資料並送出報名';

                const modal = document.getElementById('successModal');

                submitBtn.innerHTML = '<i class="fa-solid fa-paper-plane text-xl sm:text-2xl"></i> 確認資料並送出報名';
